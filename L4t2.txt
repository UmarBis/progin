#include <sys/socket.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <iostream>
#include <string>

constexpr int PORT = 60002;

int main() {
    int sock = socket(AF_INET, SOCK_STREAM, 0);

    std::string ip;
    std::cout << "Server IP: ";
    std::cin >> ip;

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(PORT);
    inet_pton(AF_INET, ip.c_str(), &addr.sin_addr);

    if (connect(sock, (sockaddr*)&addr, sizeof(addr)) < 0) {
        std::cerr << "Connection failed\n";
        return 1;
    }

    std::cout << "Connected! Guess number (1-100)\n";

    std::string msg;
    char buffer[128];

    while (true) {
        std::cout << "Your guess: ";
        std::cin >> msg;

        send(sock, msg.c_str(), msg.size(), 0);

        int n = recv(sock, buffer, sizeof(buffer), 0);
        if (n <= 0) break;

        buffer[n] = '\0';
        std::cout << "Server: " << buffer;

        if (std::string(buffer).find("correct") != std::string::npos)
            break;
    }

    close(sock);
}
